    string s;
    cin >> s;
    int i = 0, j = (int)s.size() - 1;
    bool ok = true;
    while (i < j) {

----------------------------------------------------------------

#include <iostream>
using namespace std;
int main() {
    int n, t, a[1000];
    cin >> n >> t;
    for (int i = 0; i < n; i++) cin >> a[i];
    int lo = 0, hi = n - 1, ans = -1;
    while (lo <= hi) {
        int mid = (lo + hi) / 2;
        if (a[mid] == t) { ans = mid; break; }
        if (a[mid] < t) lo = mid + 1; else hi = mid - 1;
    }
    cout << ans << endl;
    return 0;
}

    for (int i = 0; i < n; i++) {
        int x;

        else { cur->next = b; b = b->next; }
        cur = cur->next;
    }
    cur->next = a ? a : b;
    return dummy.next;
}

        for (int j = 0; j < m; j++) { cin >> v[i].score[j]; v[i].sum += v[i].score[j]; }
        v[i].idx = i;
    }
    string cmd;
    cin >> cmd;
    if (cmd == "total") {
        stable_sort(v.begin(), v.end(), [](auto& a, auto& b) { return a.sum > b.sum; });
        for (auto& x : v) cout << x.name << " " << x.sum << endl;
    } else {
        int k;
        cin >> k;
        double s = 0;
        for (auto& x : v) s += x.score[k - 1];
        printf("%.1f\n", s / n);
    }
    return 0;
}

----------------------------------------------------------------
第98题：简单文本行编辑器
----------------------------------------------------------------

#include <iostream>
#include <string>
#include <vector>
using namespace std;
int main() {
    int m;
    cin >> m;
    vector<string> lines;
    while (m--) {
        string cmd, txt;
        cin >> cmd;
        if (cmd == "ADD") {
            cin >> txt;
            lines.push_back(txt);
        } else if (cmd == "INSERT") {
            int pos;
            cin >> pos >> txt;
            if (pos >= 1 && pos <= (int)lines.size() + 1)
                lines.insert(lines.begin() + pos - 1, txt);
        } else if (cmd == "DEL") {
            int pos;
            cin >> pos;
            if (pos >= 1 && pos <= (int)lines.size())
                lines.erase(lines.begin() + pos - 1);
        } else if (cmd == "PRINT") {
            if (lines.empty()) cout << "EMPTY" << endl;
            else for (auto& s : lines) cout << s << endl;
        }
    }
    return 0;
}

第99题：日历计算
----------------------------------------------------------------

#include <iostream>
using namespace std;
bool isLeap(int y) {
    return (y % 400 == 0) || (y % 4 == 0 && y % 100 != 0);
}
int main() {
    int y, m;
    cin >> y >> m;
    int days[] = {0, 31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
    int total = 0;
    for (int i = 1900; i < y; i++) total += isLeap(i) ? 366 : 365;
    for (int i = 1; i < m; i++) total += days[i] + (i == 2 && isLeap(y) ? 1 : 0);
    int dm = days[m] + (m == 2 && isLeap(y) ? 1 : 0);
    int wd = (total + 1) % 7;  // 1900-01-01 是星期一(1)
    cout << dm << " " << wd << endl;
    return 0;
}

第100题：股票最大利润
----------------------------------------------------------------

#include <iostream>
using namespace std;
int main() {
    int n, x;
    cin >> n;
    int minP = 1e9, ans = 0;
    for (int i = 0; i < n; i++) {
        cin >> x;
        ans = max(ans, x - minP);
        minP = min(minP, x);
    }
    cout << ans << endl;
    return 0;
}
